type backend struct {mu sync.Mutex;overrides map[string]interface{};state string;sent int;tx *types.Transaction;failSend bool;autoReceipt bool}
func(b *backend)ServeHTTP(w http.ResponseWriter,r *http.Request){
 b.mu.Lock();defer b.mu.Unlock()
 var q struct{Method string `json:"method"`;Params []json.RawMessage `json:"params"`};json.NewDecoder(r.Body).Decode(&q)
 var value interface{}
 switch q.Method{
 case "net_version":value="70088"
 case "eth_getBlockByNumber":value=map[string]interface{}{"hash":genesisHash}
 case "eth_syncing":value=false
 case "net_peerCount":value="0x2"
 case "eth_getCode":value="0x"
 case "eth_getBalance":value="0x56bc75e2d63100000"
 case "eth_getTransactionCount":value="0x2a"
 case "eth_sendRawTransaction":
  b.sent++;var raw string;json.Unmarshal(q.Params[0],&raw)
  data,_:=hexutil.Decode(raw);var tx types.Transaction
  if rlp.DecodeBytes(data,&tx)!=nil{panic("fixture RLP")};b.tx=&tx
  saved,e:=ioutil.ReadFile(b.state);var j journal
  if e!=nil||json.Unmarshal(saved,&j)!=nil||j.Hash!=tx.Hash().Hex()||j.Settled{panic("signed journal not durable before send")}
  if b.failSend{json.NewEncoder(w).Encode(map[string]interface{}{"id":1,"result":nil,"error":map[string]interface{}{"message":"fixture ambiguous failure"}});return}
  value=tx.Hash().Hex()
  if b.autoReceipt{b.overrides["eth_getTransactionReceipt"]=map[string]interface{}{"transactionHash":tx.Hash().Hex(),"blockHash":genesisHash,"blockNumber":"0x1","status":"0x1"}}
 case "eth_getTransactionReceipt":value=nil
 default:panic("unapproved upstream method: "+q.Method)
 }
 if x,ok:=b.overrides[q.Method];ok{value=x}
 json.NewEncoder(w).Encode(map[string]interface{}{"jsonrpc":"2.0","id":1,"result":value,"error":nil})
}
func fixture(t *testing.T)(*signer,*backend){
 key,e:=crypto.HexToECDSA(strings.Repeat("0",63)+"1");if e!=nil{t.Fatal(e)}
 dir:=t.TempDir();os.Chmod(dir,0700)
 b:=&backend{overrides:map[string]interface{}{},state:filepath.Join(dir,"journal.json")}
 h:=httptest.NewServer(b);t.Cleanup(h.Close)
 s,e:=newSigner(config{Address:crypto.PubkeyToAddress(key.PublicKey).Hex(),State:b.state,Upstream:h.URL,PaymentsEnabled:true},key);if e!=nil{t.Fatal(e)}
 return s,b
}
func params(s *signer)json.RawMessage{
 raw,_:=json.Marshal([]map[string]string{{"from":s.address.Hex(),"to":"0x1111111111111111111111111111111111111111","value":"0xde0b6b3a7640000","gas":"0x5208","gasPrice":"0x3b9aca00"}});return raw
}
func TestValidTransferIsEIP155AndJournaledBeforeBroadcast(t *testing.T){
 s,b:=fixture(t);v,e:=s.send(params(s));if e!=nil{t.Fatal(e)}
 b.mu.Lock();defer b.mu.Unlock()
 if b.sent!=1||v!=b.tx.Hash().Hex()||b.tx.ChainId().Int64()!=chainID||b.tx.Nonce()!=42||b.tx.Gas()!=21000||len(b.tx.Data())!=0{t.Fatal("payout policy differs")}
 if b.tx.To().Hex()!="0x1111111111111111111111111111111111111111"||b.tx.Value().String()!="1000000000000000000"{t.Fatal("wrong recipient or amount")}
 from,e:=types.Sender(types.NewEIP155Signer(b.tx.ChainId()),b.tx);if e!=nil||from!=s.address{t.Fatal("invalid sender signature")}
 if i,e:=os.Stat(b.state);e!=nil||i.Mode().Perm()!=0600{t.Fatal("journal permissions")}
}

func TestCanonicalSuccessfulReceiptAllowsNextAdvancedNonce(t *testing.T){
