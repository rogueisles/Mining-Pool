def read_json(path,uid=0,modes=(0o600,0o640,0o644),limit=65536):
 fd=os.open(path,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK)
 with os.fdopen(fd,'rb') as f:
  i=os.fstat(f.fileno())
  if not stat.S_ISREG(i.st_mode) or i.st_uid!=uid or i.st_nlink!=1 or stat.S_IMODE(i.st_mode) not in modes or i.st_size>limit:raise ValueError('Health source metadata differs')
  return json.load(f)
def rpc(url,method,params,auth=None):
 allowed={'net_version','net_peerCount','eth_getBlockByNumber','eth_blockNumber','eth_syncing','getblockchaininfo','getnetworkinfo','getblockhash'}
 if method not in allowed:raise ValueError('Only read-only health methods are permitted')
 q=urllib.request.Request(url,json.dumps(dict(jsonrpc='2.0',id=1,method=method,params=params)).encode(),{'Content-Type':'application/json'})
 if auth:q.add_header('Authorization','Basic '+base64.b64encode((auth['username']+':'+auth['password']).encode()).decode())
 with HTTP.open(q,timeout=3) as r:raw=r.read(1024*1024+1)
 if len(raw)>1024*1024:raise ValueError('Health response too large')
 v=json.loads(raw)
 if v.get('id')!=1 or v.get('error'):raise ValueError('Health RPC unavailable')
 return v['result']
def batch(url,calls,auth=None):
 allowed={'net_version','eth_getBlockByNumber','eth_blockNumber','getblockchaininfo','getnetworkinfo','getblockhash'}
 if any(m not in allowed for m,p in calls):raise ValueError('Read-only health batch required')
 body=[dict(jsonrpc='2.0',id=i+1,method=m,params=p) for i,(m,p) in enumerate(calls)]
 q=urllib.request.Request(url,json.dumps(body).encode(),{'Content-Type':'application/json'})
 if auth:q.add_header('Authorization','Basic '+base64.b64encode((auth['username']+':'+auth['password']).encode()).decode())
 with HTTP.open(q,timeout=6) as response:raw=response.read(1024*1024+1)
 if len(raw)>1024*1024:raise ValueError('Health batch too large')
 rows=json.loads(raw)
 if not isinstance(rows,list) or len(rows)!=len(calls) or {v.get('id') for v in rows}!=set(range(1,len(calls)+1)) or any(v.get('error') for v in rows):raise ValueError('Health batch unavailable')
 result={v['id']:v['result'] for v in rows};return [result[i+1] for i in range(len(calls))]

def write_public(data):

  with os.fdopen(fd,'wb') as f:os.fchmod(f.fileno(),0o644);f.write((json.dumps(data,indent=2)+'\n').encode());f.flush();os.fsync(f.fileno())
  os.replace(name,OUTPUT)
 finally:
  if os.path.lexists(name):os.unlink(name)
